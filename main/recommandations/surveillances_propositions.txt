@shared_task
def run_surveillance_for_frequency(frequency):
    """
    Tâche Celery pour exécuter la surveillance pour une fréquence donnée.
    'frequency' peut être 'quotidienne', 'hebdomadaire', 'mensuelle'.
    """
    print(f"--- Lancement de la surveillance {frequency} ---")

    # 1. Récupérer les portefeuilles concernés
    portefeuilles_a_verifier = Portefeuille.objects.filter(frequence_alertes=frequency)

    for portefeuille in portefeuilles_a_verifier:
        print(f"Vérification du portefeuille : {portefeuille.nom}")

        # Date de la dernière vérification pour ce portefeuille
        last_check_date = portefeuille.derniere_verification

        # 2. Récupérer les acheteurs liés à ce portefeuille
        acheteurs_surveilles = portefeuille.portefeuilleclient_set.all().select_related('acheteur')

        # 3. Récupérer les éléments de surveillance actifs pour ce portefeuille
        elements_actifs = portefeuille.elements_surveillance_actifs.all()

        for item_portefeuille_client in acheteurs_surveilles:
            acheteur = item_portefeuille_client.acheteur
            print(f"  -> Surveillance de l'acheteur : {acheteur.nom}")

            for element in elements_actifs:
                # 4. Trouver la fonction de vérification correspondante dans notre "Registry"
                checker_function = CHECKER_REGISTRY.get(element.code_interne)

                if checker_function:
                    # 5. Exécuter la logique de vérification !
                    checker_function(
                        portefeuille=portefeuille,
                        acheteur=acheteur,
                        element_surveille=element,
                        last_check_date=last_check_date
                    )

        # 6. Mettre à jour la date de dernière vérification pour ne pas re-scanner
        portefeuille.derniere_verification = timezone.now()
        portefeuille.save(update_fields=['derniere_verification'])

    return f"Surveillance {frequency} terminée avec succès pour {portefeuilles_a_verifier.count()} portefeuilles."

@receiver(post_save, sender=CredendoCommande)
def envoyer_email_aux_analystes_et_validateurs(sender, instance, created, **kwargs):
    if created:  # Vérifie si la commande vient d'être créée
        pays_commande = instance.pays  # Récupère le pays de la commande

        # Rechercher les utilisateurs Analyste et Validateur avec le même pays
        utilisateurs_cibles = CustomUser.objects.filter(
            role__in=['Analyste', 'Validateur'],
            pays__code=pays_commande  # Si "pays" est une ForeignKey
        ) | CustomUser.objects.filter(
            role__in=['Analyste', 'Validateur'],
            pays=pays_commande  # Si "pays" est stocké en texte
        )

        if utilisateurs_cibles.exists():
            emails = utilisateurs_cibles.values_list('email', flat=True)  # Récupérer leurs emails

            sujet = "Nouvelle commande enregistrée"
            message = f"Une nouvelle commande a été enregistrée pour {pays_commande}.\n\n"
            message += f"Référence: {instance.reference}\nNom: {instance.nom}\nMontant: {instance.montant} {instance.devise}\n\n"
            message += "Connectez-vous pour plus de détails."

            send_mail(
                sujet,
                message,
                settings.DEFAULT_FROM_EMAIL,
                list(emails),
                fail_silently=False,
            )



def send_ws_notification(user, type_notif, message, commande=None):
    """Envoie une notification en temps réel via WebSockets."""
    channel_layer = get_channel_layer()
    async_to_sync(channel_layer.group_send)(
        f"user_{user.id}",
        {
            "type": "send_notification",
            "message": message,
            "type": type_notif,
            "commande": commande.notre_ref if commande else None,
        }
    )

@shared_task
def nettoyer_historique():
    from .models import HistoriqueDonnees, EvenementSurveillance

    limite_date = timezone.now() - timedelta(days=180)

    anciens_historiques = HistoriqueDonnees.objects.filter(date_capture__lt=limite_date)
    count_historiques = anciens_historiques.count()
    anciens_historiques.delete()

    limite_evenements = timezone.now() - timedelta(days=90)
    anciens_evenements = EvenementSurveillance.objects.filter(
        traite=True,
        date_traitement__lt=limite_evenements
    )
    count_evenements = anciens_evenements.count()
    anciens_evenements.delete()

    logger.info(f"Nettoyage terminé: {count_historiques} historiques et {count_evenements} événements supprimés")
